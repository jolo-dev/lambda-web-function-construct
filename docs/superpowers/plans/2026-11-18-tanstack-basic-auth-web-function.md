export const DEMO_USER = {
  email: "demo@example.com",
  password: "demo-password",
} as const;

const digest = (value: string) => createHash("sha256").update(value).digest("hex");
const DEMO_PASSWORD_HASH = digest(DEMO_USER.password);

/** Demo-only comparison mirroring the upstream starter; never use for real credentials. */
export function demoCredentialsMatch(email: string, password: string): boolean {
  return email === DEMO_USER.email && digest(password) === DEMO_PASSWORD_HASH;
}

```ts
import { getIronSession } from "iron-session";
import { getCookie, setCookie } from "@tanstack/react-start/server";

export interface SessionUser {
  userEmail?: string;
}

export function getAppSession() {
  const password = process.env.SESSION_PASSWORD;
  if (!password || password.length < 32) {
    throw new Error("Set SESSION_PASSWORD to at least 32 characters");
  }
  return getIronSession<SessionUser>(
    { read: getCookie, write: setCookie },
    {
      cookieName: "app-session",
      password,
      ttl: 7 * 24 * 60 * 60,
      cookieOptions: {
        httpOnly: true,
        sameSite: "lax",
        secure: process.env.NODE_ENV === "production",
        path: "/",
      },
    },
  );
}
```

async function authPost(path: string, body: unknown, cookie?: string) {
  return fetch(new URL(path, baseUrl()), {
    method: "POST",
    headers: { "content-type": "application/json", ...(cookie ? { cookie } : {}) },
    body: JSON.stringify(body),
    redirect: "manual",
    signal: AbortSignal.timeout(10_000),
  });
}

async function getWithCookie(path: string, cookie?: string) {
  const response = await fetch(new URL(path, baseUrl()), {
    headers: cookie ? { cookie } : {},
    redirect: "manual",
    signal: AbortSignal.timeout(10_000),
  });
  return { status: response.status, html: await response.text() };
}

  const anonymous = await getWithCookie("/profile");
  expect(anonymous.html).toContain("Sign in");

  const login = await authPost("/api/login", {
    email: "demo@example.com",
    password: "demo-password",
  });
  const cookie = (login.headers.get("set-cookie") ?? "").split(";")[0] ?? "";
  const session = await fetch(new URL("/api/session", baseUrl()), { headers: { cookie } });
  expect(await session.json()).toEqual({ email: "demo@example.com" });

  const authed = await getWithCookie("/profile", cookie);
  expect(authed.html).toContain("demo@example.com");

  const logout = await fetch(new URL("/api/logout", baseUrl()), {
    method: "POST",
    headers: { cookie },
    redirect: "manual",
  });
  expect(logout.status).toBe(200);
  const logoutCookie = logout.headers.get("set-cookie") ?? "";
  expect(logoutCookie).toContain("app-session=");
  const cleared = logoutCookie.split(";")[0] ?? "";
  const afterLogout = await getWithCookie("/profile", cleared);
  expect(afterLogout.html).toContain("Sign in");

  // iron-session is stateless: destroy() clears the browser cookie but cannot revoke an old copy.
  const replay = await getWithCookie("/profile", cookie);
  expect(replay.html).toContain("demo@example.com");
});

import { createFileRoute } from "@tanstack/react-router";
import { json } from "@tanstack/react-start";
import { demoCredentialsMatch, DEMO_USER } from "../utils/demo-user";
import { getAppSession } from "../utils/session";

export const Route = createFileRoute("/api/login")({
  server: {
    handlers: {
      POST: async ({ request }) => {
        const body = await request.json().catch(() => ({})) as { email?: unknown; password?: unknown };
        const email = typeof body.email === "string" ? body.email : "";
        const password = typeof body.password === "string" ? body.password : "";
        if (!demoCredentialsMatch(email, password)) {
          return json({ error: true, message: "Invalid email or password" }, { status: 401 });
        }
        const session = await getAppSession();
        session.userEmail = DEMO_USER.email;
        await session.save();
        return json({ email: DEMO_USER.email });
      },
    },
  },
});

It must render the literal text `Sign in`, show the demo credentials, submit JSON via `fetch("/api/login", { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify({ email, password }) })`, and render the returned error message on failure. Keep it plain CSS (no Tailwind):

```tsx
import { useState } from "react";
import { DEMO_USER } from "../utils/demo-user";

export function LoginPanel() {
  const [email, setEmail] = useState(DEMO_USER.email);
  const [password, setPassword] = useState(DEMO_USER.password);
  const [error, setError] = useState<string | null>(null);
  async function submit(e: React.FormEvent) {
    e.preventDefault();
    setError(null);
    const response = await fetch("/api/login", {
      method: "POST",
      headers: { "content-type": "application/json" },
      body: JSON.stringify({ email, password }),
    });
    if (!response.ok) {
      const body = await response.json().catch(() => ({})) as { message?: string };
      setError(body.message ?? "Sign in failed");
      return;
    }
    window.location.assign("/profile");
  }
  return (
    <main>
      <h1>Sign in</h1>
      <p>Demo credentials: {DEMO_USER.email} / {DEMO_USER.password}</p>
      <form onSubmit={submit}>
        <label htmlFor="email">Email</label>
        <input id="email" value={email} onChange={(e) => setEmail(e.target.value)} />
        <label htmlFor="password">Password</label>
        <input id="password" type="password" value={password} onChange={(e) => setPassword(e.target.value)} />
        <button type="submit">Sign in</button>
      </form>
      {error ? <p id="login-error">{error}</p> : null}
    </main>
  );
}
```
